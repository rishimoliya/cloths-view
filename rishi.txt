        // --- 5. FILTERING & SEARCH SYSTEM ---
        function applyFilters() {
            const searchVal = document.getElementById('searchInput').value.toLowerCase();
            const catVal = document.getElementById('categoryFilter').value;
            const ageVal = document.getElementById('ageFilter').value;
            const matVal = document.getElementById('materialFilter').value;
            const maxPrice = parseFloat(document.getElementById('priceFilter').value);
            const sortVal = document.getElementById('sortOrder').value;

            document.getElementById('priceDisplay').innerText = `₹${maxPrice}`;

            let filtered = products.filter(p => {
                const matchesSearch = p.name.toLowerCase().includes(searchVal) || p.material.toLowerCase().includes(searchVal);
                const matchesCat = catVal === 'all' || p.category === catVal;
                const matchesAge = ageVal === 'all' || p.ageGroup === ageVal;
                const matchesMat = matVal === 'all' || p.material === matVal;
                const matchesPrice = p.price <= maxPrice;

                return matchesSearch && matchesCat && matchesAge && matchesMat && matchesPrice;
            });

            // Sorting
            if (sortVal === 'price-low') {
                filtered.sort((a, b) => a.price - b.price);
            } else if (sortVal === 'price-high') {
                filtered.sort((a, b) => b.price - a.price);
            } else if (sortVal === 'rating') {
                filtered.sort((a, b) => b.rating - a.rating);
            }

            renderProducts(filtered);
        }

        function processAiQuery() {
            const input = document.getElementById('aiInput').value.toLowerCase().trim();
            const resultsContainer = document.getElementById('aiResults');

            if (!input) return;

            // Simple parser rules
            let matchedProducts = products.filter(p => {
                let score = 0;
                
                // Age rules
                if ((input.includes('baby') || input.includes('6 month') || input.includes('newborn')) && (p.ageGroup === '0-3m' || p.ageGroup === '3-12m')) score += 2;
                if (input.includes('19') && p.ageGroup === '19') score += 3;
                if (input.includes('22') && p.ageGroup === '22') score += 3;
                if ((input.includes('58') || input.includes('senior') || input.includes('elder')) && p.ageGroup === '58') score += 3;

                // Material rules
                if (input.includes('muslin') && p.material === 'Muslin') score += 2;
                if (input.includes('cotton') && p.material.includes('Cotton')) score += 2;
                if (input.includes('linen') && p.material === 'Linen') score += 2;
                if (input.includes('bamboo') && p.material.includes('Bamboo')) score += 2;

                // Price rules
                if (input.includes('under 1000') || input.includes('under ₹1000')) {
                    if (p.price <= 1000) score += 2;
                }
                if (input.includes('under 800') || input.includes('under ₹800')) {
                    if (p.price <= 800) score += 2;
                }

                // General keywords
                if (input.includes('gift') || input.includes('set')) {
                    if (p.category === 'sets' || p.category === 'blanket') score += 1;
                }

                return score > 0;
            });

            // Fallback if no specific match
            if (matchedProducts.length === 0) {
                matchedProducts = products.slice(0, 3);
            }

            resultsContainer.innerHTML = matchedProducts.slice(0, 3).map(p => `
                <div class="flex items-center gap-3 p-2 bg-brand-warmWhite rounded-2xl border border-brand-sand/50">
                    <img src="${p.image}" class="w-12 h-12 rounded-xl object-cover">
                    <div class="flex-grow">
                        <p class="font-bold text-xs text-brand-charcoal">${p.name}</p>
                        <p class="text-[10px] text-brand-muted">${p.ageLabel} • ${p.material} • ₹${p.price}</p>
                    </div>
                    <button onclick="addToCart(${p.id}, '${p.sizes[0]}'); toggleAiModal(false);" class="px-3 py-1.5 bg-brand-brown text-white text-[11px] font-bold rounded-xl hover:bg-brand-brownDark transition-colors">
                        Add
                    </button>
                </div>
            `).join('');
        }

        // --- 8. SHOPPING CART LOGIC ---
        function addToCart(productId, size) {
            const product = products.find(p => p.id === productId);
            if (!product) return;

            const existingIndex = cart.findIndex(item => item.id === productId && item.selectedSize === size);

            if (existingIndex > -1) {
                cart[existingIndex].qty += 1;
            } else {
                cart.push({
                    id: product.id,
                    name: product.name,
                    price: product.price,
                    image: product.image,
                    selectedSize: size,
                    qty: 1
                });
            }

            saveCart();
            updateCartDrawer();
            showToast(`Added "${product.name}" to cart!`);
        }

        function updateCartQty(index, delta) {
            cart[index].qty += delta;
            if (cart[index].qty <= 0) {
                cart.splice(index, 1);
            }
            saveCart();
            updateCartDrawer();
        }

        function saveCart() {
            localStorage.setItem('minime_cart', JSON.stringify(cart));
            updateCartBadge();
        }

        function updateCartBadge() {
            const totalItems = cart.reduce((sum, item) => sum + item.qty, 0);
            document.getElementById('cartBadge').innerText = totalItems;
            document.getElementById('cartItemCount').innerText = totalItems;
        }

        function updateCartDrawer() {
            const container = document.getElementById('cartItemsContainer');
            
            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-12 space-y-3">
                        <div class="text-3xl text-brand-sand">🛍️</div>
                        <p class="text-xs font-bold text-brand-charcoal">Your cart is currently empty.</p>
                        <p class="text-[11px] text-brand-muted">Discover soft organic garments from our catalog.</p>
                    </div>
                `;
            } else {
                container.innerHTML = cart.map((item, index) => `
                    <div class="flex items-center gap-3 p-3 bg-brand-warmWhite rounded-2xl border border-brand-sand/50">
                        <img src="${item.image}" class="w-14 h-14 object-cover rounded-xl">
                        <div class="flex-grow">
                            <h4 class="font-bold text-xs text-brand-charcoal">${item.name}</h4>
                            <p class="text-[10px] text-brand-muted">Size: ${item.selectedSize}</p>
                            <span class="font-bold text-xs text-brand-brown">₹${item.price}</span>
                        </div>
                        <div class="flex items-center gap-2 border border-brand-sand rounded-xl bg-white px-2 py-1">
                            <button onclick="updateCartQty(${index}, -1)" class="text-xs font-bold text-brand-muted px-1">-</button>
                            <span class="text-xs font-bold">${item.qty}</span>
                            <button onclick="updateCartQty(${index}, 1)" class="text-xs font-bold text-brand-muted px-1">+</button>
                        </div>
                    </div>
                `).join('');
            }

            // Price Calculations
            const subtotal = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
            const discountAmt = Math.round(subtotal * appliedDiscount);
            const shipping = (subtotal > 999 || subtotal === 0) ? 0 : 99;
            const grandTotal = Math.max(0, subtotal - discountAmt + shipping);

            document.getElementById('cartSubtotal').innerText = `₹${subtotal}`;
            document.getElementById('cartDiscount').innerText = `-₹${discountAmt}`;
            document.getElementById('cartShipping').innerText = shipping === 0 ? 'FREE' : `₹${shipping}`;
            document.getElementById('cartTotal').innerText = `₹${grandTotal}`;
            document.getElementById('checkoutTotalAmount').innerText = `₹${grandTotal}`;
        }

        function applyCoupon() {
            const code = document.getElementById('couponInput').value.trim().toUpperCase();
            const msg = document.getElementById('couponMsg');

            if (code === 'MINI10') {
                appliedDiscount = 0.10;
                msg.innerText = "✨ Promo code 'MINI10' applied (10% OFF)!";
                msg.classList.remove('hidden');
                showToast("10% discount code applied!");
            } else {
                appliedDiscount = 0;
                msg.innerText = "Invalid code. Try 'MINI10'.";
                msg.classList.remove('hidden');
                msg.classList.add('text-red-500');
            }
            updateCartDrawer();
        }

            }
        }

                showToast("Your cart is empty!");
                return;
            }
            toggleCartDrawer(false);
            document.getElementById('checkoutModal').classList.remove('hidden');
        }

        function toggleCheckoutModal(show) {
            const modal = document.getElementById('checkoutModal');
            if (show) modal.classList.remove('hidden');
            else modal.classList.add('hidden');
        }

        function processOrder(e) {
            e.preventDefault();

            const name = document.getElementById('custName').value;
            const email = document.getElementById('custEmail').value;
            const phone = document.getElementById('custPhone').value;
            const pin = document.getElementById('custPin').value;
            const address = document.getElementById('custAddress').value;

            const orderId = `#MINI-${Math.floor(10000 + Math.random() * 90000)}`;
            const dateStr = new Date().toLocaleDateString('en-IN', { dateStyle: 'medium' });

            // Populate Invoice
            document.getElementById('invOrderId').innerText = orderId;
            document.getElementById('invDate').innerText = dateStr;
            document.getElementById('invCustName').innerText = name;
            document.getElementById('invCustEmail').innerText = email;
            document.getElementById('invCustPhone').innerText = phone;
            document.getElementById('invCustAddress').innerText = `${address}, PIN: ${pin}`;

            // Invoice Table
            const subtotal = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
            const discountAmt = Math.round(subtotal * appliedDiscount);
            const total = Math.max(0, subtotal - discountAmt);

            document.getElementById('invItemsTable').innerHTML = cart.map(item => `
                <tr class="border-b border-stone-100">
                    <td class="py-2 font-medium text-stone-800">${item.name}</td>
                    <td class="py-2 text-stone-500">${item.selectedSize}</td>
                    <td class="py-2 text-stone-500">${item.qty}</td>
                    <td class="py-2 text-right font-semibold text-stone-800">₹${item.price * item.qty}</td>
                </tr>
            `).join('');

            document.getElementById('invSubtotal').innerText = `₹${subtotal}`;
            document.getElementById('invDiscount').innerText = `-₹${discountAmt}`;
            document.getElementById('invTotal').innerText = `₹${total}`;

            // Reset Cart
            cart = [];
            saveCart();

            toggleCheckoutModal(false);
            document.getElementById('invoiceModal').classList.remove('hidden');
            showToast("Order placed successfully!");
        }

        function closeInvoiceModal() {
            document.getElementById('invoiceModal').classList.add('hidden');
        }

        // --- 10. WISHLIST SYSTEM ---
        function toggleWishlist(id) {
            const index = wishlist.indexOf(id);
            if (index > -1) {
                wishlist.splice(index, 1);
                showToast("Item removed from Wishlist");
            } else {
                wishlist.push(id);
                showToast("Item saved to Wishlist ❤️");
            }
            localStorage.setItem('minime_wishlist', JSON.stringify(wishlist));
            updateWishlistBadge();
            applyFilters();
        }

        function updateWishlistBadge() {
            const badge = document.getElementById('wishlistBadge');
            badge.innerText = wishlist.length;
            if (wishlist.length > 0) badge.classList.remove('hidden');
            else badge.classList.add('hidden');
        }

        function toggleWishlistModal(show) {
            const modal = document.getElementById('wishlistModal');
            if (show) {
                renderWishlistModal();
                modal.classList.remove('hidden');
            } else {
                modal.classList.add('hidden');
            }
        }

        function renderWishlistModal() {
            const container = document.getElementById('wishlistContainer');
            const items = products.filter(p => wishlist.includes(p.id));

            if (items.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-8 text-xs text-brand-muted">
                        No saved items in your wishlist yet.
                    </div>
                `;
                return;
            }

            container.innerHTML = items.map(p => `
                <div class="flex items-center gap-3 p-3 bg-brand-warmWhite rounded-2xl border border-brand-sand/50">
                    <img src="${p.image}" class="w-12 h-12 rounded-xl object-cover">
                    <div class="flex-grow">
                        <p class="font-bold text-xs text-brand-charcoal">${p.name}</p>
                        <p class="text-[10px] text-brand-muted">₹${p.price}</p>
                    </div>
                    <button onclick="addToCart(${p.id}, '${p.sizes[0]}'); toggleWishlistModal(false);" class="px-3 py-1.5 bg-brand-brown text-white text-xs font-bold rounded-xl">
                        Move to Cart
                    </button>
                </div>
            `).join('');
        }

        // --- 11. TOAST SYSTEM & UTILS ---
        function showToast(message) {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            toast.className = "bg-brand-charcoal text-white text-xs font-medium px-4 py-3 rounded-2xl shadow-xl flex items-center gap-2 transition-all duration-300 transform translate-y-2 pointer-events-auto";
            toast.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-400"></i> ${message}`;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-4');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
